cmake_minimum_required(VERSION 3.16)
project(tau_spin_study LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED ENV{TAUSPIN_STACK})
  message(FATAL_ERROR "Source env.sh first")
endif()
set(STACK $ENV{TAUSPIN_STACK})
if(DEFINED ENV{TAUOLA_ROOT})
  set(TAUOLA $ENV{TAUOLA_ROOT})
  include_directories(BEFORE ${TAUOLA}/include)
  link_directories(BEFORE ${TAUOLA}/lib)
else()
  set(TAUOLA ${STACK})
endif()
include_directories(${STACK}/include)
link_directories(${STACK}/lib ${STACK}/lib64)
set(CMAKE_BUILD_RPATH "${TAUOLA}/lib;${STACK}/lib;${STACK}/lib64")
add_executable(tau-spin src/main.cc)
file(READ "${TAUOLA}/include/TauSpinner/tau_reweight_lib.h" SPINNER_HEADER)
if(SPINNER_HEADER MATCHES "initialize_gamagama")
  target_compile_definitions(tau-spin PRIVATE SPINNER_PHOTON_API=1)
  add_executable(check-tables src/check_tables.cc)
  target_link_libraries(check-tables PRIVATE TauolaTauSpinner TauolaCxxInterface TauolaFortran LHAPDF)
endif()
target_link_libraries(tau-spin PRIVATE pythia8 HepMC3 TauolaTauSpinner TauolaCxxInterface TauolaFortran LHAPDF)
add_executable(readcheck src/readcheck.cc)
target_link_libraries(readcheck PRIVATE HepMC3)
add_executable(check-collinear tests/collinear.cc)
